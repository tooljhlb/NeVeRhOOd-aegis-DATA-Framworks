cmake_minimum_required(VERSION 3.24)
project(aegis_data LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(aegis_reference_test native/tests/reference_test.cpp)
target_compile_options(aegis_reference_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME reference COMMAND aegis_reference_test)
option(AEGIS_ENABLE_CUDA "Compile and test CUDA flow primitive; requires a GPU to run" OFF)
if(AEGIS_ENABLE_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  add_library(aegis_flow STATIC native/cuda/flow.cu)
  add_executable(aegis_cuda_test native/tests/cuda_test.cu)
  target_link_libraries(aegis_cuda_test PRIVATE aegis_flow)
  add_test(NAME cuda_flow COMMAND aegis_cuda_test)
endif()
